"use strict";
const $=id=>document.getElementById(id);
let owner="";
let busy=false;
const setStatus=text=>$("status").textContent=text;
async function call(url,body){
  const options=body===undefined?{cache:"no-store"}:{method:"POST",headers:{"Content-Type":"application/json"},body:JSON.stringify(body),cache:"no-store"};
  const response=await fetch(url,options);
  const data=await response.json();
  if(!response.ok||!data.ok)throw new Error(data.error||"接続できませんでした");
  return data;
}
const percent=n=>$("progress").style.width=Math.max(0,Math.min(100,n))+"%";
function prettyDate(value){
  if(!value)return "—";
  const date=new Date(value);return Number.isNaN(date.getTime())?"—":date.toLocaleString("ja-JP");
}
function render(data){
  const items=Array.isArray(data?.items)?data.items:[];
  $("count").textContent=items.length;
  $("updated").textContent="前回取得："+prettyDate(data?.updatedAt);
  $("articleCount").textContent="記事："+String(data?.articleCount??"—");
  $("details").textContent=data?.failedArticles?String(data.failedArticles)+"記事は判定保留です":"";
  const root=$("comments");root.replaceChildren();
  if(!items.length){
    const empty=document.createElement("div");empty.className="panel";
    empty.textContent=data?.updatedAt?"未対応コメントはありません🌙":"noteユーザー名を登録して、初回取得してください。";
    root.append(empty);return;
  }
  for(const item of items){
    const card=document.createElement("article");card.className="comment";
    const who=document.createElement("div");who.className="who";who.textContent=item.authorName||item.authorUrlname||"ゲスト";
    const date=document.createElement("div");date.className="date";date.textContent=prettyDate(item.publishedAt);
    const title=document.createElement("h3");title.textContent=item.articleTitle||"記事";
    const body=document.createElement("p");body.textContent=item.body||"";
    const actions=document.createElement("div");actions.className="actions";
    const link=document.createElement("a");link.href=item.articleUrl;link.target="_blank";link.rel="noopener noreferrer";link.textContent="noteで開く ↗";
    const copy=document.createElement("button");copy.textContent="ChatGPT用にコピー";
    copy.addEventListener("click",async()=>{
      const prompt=["次のnoteコメントへの返信案を日本語で作成してください。","記事："+(item.articleTitle||""),
        "URL："+(item.articleUrl||""),"投稿者："+(item.authorName||""),
        "コメント："+(item.body||""),"先に返信した内容：" +(item.rootBody||""),
        "相手の内容に具体的に触れ、適切な長さで丁寧かつ自然に返信してください。"].join("\n");
      try{await navigator.clipboard.writeText(prompt);copy.textContent="コピーしました ✓";}
      catch{copy.textContent="コピーできませんでした";}
    });
    actions.append(copy,link);card.append(who,date,title,body,actions);root.append(card);
  }
}
async function initial(){
  try{
    const data=await call("/api/me");
    $("login").textContent="ログイン中："+data.email;
    owner=data.owner||"";
    $("owner").value=owner;
    $("refresh").disabled=!owner;
    render(data.latest);
    setStatus(owner?"準備完了。ボタンを押すと変更分を確認します。":"noteユーザー名を登録してください。");
  }catch(error){setStatus("認証・読み込みに失敗："+error.message);}
}
$("profileForm").addEventListener("submit",async event=>{
  event.preventDefault();
  if(busy)return;
  try{
    const data=await call("/api/profile",{owner:$("owner").value.trim()});
    owner=data.owner;$("owner").value=owner;$("refresh").disabled=false;
    render(null);setStatus("登録しました。初回取得を実行してください。");
  }catch(error){setStatus("保存できませんでした："+error.message);}
});
$("refresh").addEventListener("click",async()=>{
  if(busy||!owner)return;
  busy=true;$("refresh").disabled=true;percent(4);
  try{
    setStatus("記事一覧を確認しています…");
    let data=await call("/api/sync/start",{});
    let rounds=0;
    while(!data.done){
      if(++rounds>10000)throw new Error("走査上限に到達しました");
      const total=Number(data.candidateArticles||0),finished=Number(data.scannedArticles||0);
      percent(total?5+90*finished/total:95);
      setStatus("コメントを確認中："+finished+" / "+total+"記事");
      data=await call("/api/sync/next",{scanId:data.scanId});
    }
    render(data);percent(100);
    setStatus("差分更新が完了しました。"+(data.failedArticles?"取得失敗 "+data.failedArticles+"記事は次回再確認します。":""));
  }catch(error){setStatus("更新失敗："+error.message+"。前回の結果は保持されています。");percent(0);}
  finally{busy=false;$("refresh").disabled=false;}
});
initial();
